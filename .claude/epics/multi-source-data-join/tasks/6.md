### SSE进度推送系统
```python
@router.get("/api/join/progress/{task_id}")
async def join_progress_stream(task_id: str):
    """SSE进度推送端点"""
    async def event_generator():
        while True:
            # 从Redis获取任务状态
            status = await get_task_status_from_redis(task_id)
            if status:
                yield f"data: {json.dumps(status)}

"
            if status.get('state') in ['completed', 'failed']:
                break
            await asyncio.sleep(1)
    
    return StreamingResponse(
        event_generator(),
        media_type="text/event-stream",
        headers={
            "Cache-Control": "no-cache",
            "Connection": "keep-alive",
        }
    )

# 进度状态数据结构
class TaskProgress:
    task_id: str
    state: str  # pending/running/completed/failed
    progress_percent: float  # 0-100
    current_step: str
    estimated_remaining: int  # seconds
    memory_usage: float
    error_message: Optional[str]
```

class APIError(HTTPException):
    def __init__(self, 
                 status_code: int,
                 error_code: str,
                 message: str,
                 details: dict = None):
        self.error_code = error_code
        self.details = details or {}
        super().__init__(status_code=status_code, detail={
            "error_code": error_code,
            "message": message,
            "details": self.details
        })

# 全局异常处理
@app.exception_handler(ValidationError)
async def validation_exception_handler(request: Request, exc: ValidationError):
